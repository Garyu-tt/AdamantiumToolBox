cmake_minimum_required(VERSION 3.15)
project(mas_universal_graph_ir LANGUAGES CXX)
add_library(mas_universal_graph_ir STATIC graph_ir.cpp)
target_compile_features(mas_universal_graph_ir PUBLIC cxx_std_17)
target_include_directories(mas_universal_graph_ir PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(mas_universal_graph_ir_test ../../../tests/universal_graph_ir_native_test.cpp)
target_link_libraries(mas_universal_graph_ir_test PRIVATE mas_universal_graph_ir)
enable_testing()
add_test(NAME mas_universal_graph_ir_native COMMAND mas_universal_graph_ir_test)
