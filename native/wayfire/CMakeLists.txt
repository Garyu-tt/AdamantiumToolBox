cmake_minimum_required(VERSION 3.21)
project(MechanicusWayfireAdapter LANGUAGES CXX)
add_library(mas_wayfire_frames STATIC ipc_frames.cpp)
target_compile_features(mas_wayfire_frames PUBLIC cxx_std_17)
target_include_directories(mas_wayfire_frames PUBLIC .)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 add_library(mas_wayfire_ipc STATIC unix_client.cpp)
 target_compile_features(mas_wayfire_ipc PUBLIC cxx_std_17)
 target_link_libraries(mas_wayfire_ipc PUBLIC mas_wayfire_frames)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(mas_wayfire_frame_test tests/ipc_frames_test.cpp)
 target_link_libraries(mas_wayfire_frame_test PRIVATE mas_wayfire_frames)
 add_test(NAME wayfire_frame_codec COMMAND mas_wayfire_frame_test)
 if(TARGET mas_wayfire_ipc)
  find_package(Threads REQUIRED)
  add_executable(mas_wayfire_unix_test tests/unix_client_test.cpp)
  target_link_libraries(mas_wayfire_unix_test PRIVATE mas_wayfire_ipc Threads::Threads)
  add_test(NAME wayfire_unix_ipc COMMAND mas_wayfire_unix_test)
 endif()
endif()
