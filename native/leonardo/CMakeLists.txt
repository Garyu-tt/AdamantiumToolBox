cmake_minimum_required(VERSION 3.21)
project(MechanicusLeonardoNative LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
find_path(BOOST_JSON_HEADER boost/json.hpp REQUIRED)
add_library(mas_leonardo_vector STATIC src/boost_json.cpp src/procedural.cpp
 src/svg.cpp src/artifact.cpp)
target_compile_features(mas_leonardo_vector PUBLIC cxx_std_17)
target_include_directories(mas_leonardo_vector PUBLIC include ${BOOST_JSON_HEADER})
target_link_libraries(mas_leonardo_vector PUBLIC OpenSSL::Crypto Threads::Threads)
option(MAS_LEONARDO_SERVER "Build standalone native MAS HTTP provider" ON)
if(MAS_LEONARDO_SERVER)
 add_executable(mas-leonardo-native src/server.cpp src/server_main.cpp)
 target_include_directories(mas-leonardo-native PRIVATE ${BOOST_JSON_HEADER})
 target_link_libraries(mas-leonardo-native PRIVATE mas_leonardo_vector)
 target_compile_definitions(mas-leonardo-native PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(mas_leonardo_test tests/vector_test.cpp)
 target_link_libraries(mas_leonardo_test PRIVATE mas_leonardo_vector)
 add_test(NAME leonardo_native_vector COMMAND mas_leonardo_test)
 if(MAS_LEONARDO_SERVER)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME leonardo_native_mas_http COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/mas_http_test.py $<TARGET_FILE:mas-leonardo-native>)
  add_test(NAME leonardo_native_concurrency COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/concurrency_test.py $<TARGET_FILE:mas-leonardo-native>)
 endif()
endif()
