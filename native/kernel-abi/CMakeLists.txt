cmake_minimum_required(VERSION 3.21)
project(AdamantiumNativeHostABI LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mas_host_abi_v1 SHARED src/host_abi.cpp)
target_compile_definitions(mas_host_abi_v1 PRIVATE MAS_HOST_ABI_BUILD=1)
target_include_directories(mas_host_abi_v1 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../standards/native")
set_target_properties(mas_host_abi_v1 PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(mas_host_abi_v1 PRIVATE /W4 /WX)
else()
  target_compile_options(mas_host_abi_v1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(mas_host_c_abi_test tests/abi_smoke.c)
target_link_libraries(mas_host_c_abi_test PRIVATE mas_host_abi_v1)
enable_testing()
add_test(NAME native_host_c_abi_v1 COMMAND mas_host_c_abi_test)

# Internal headless C++ graph: never an exposed plugin authorization API.
find_package(Threads REQUIRED)
add_library(mas_native_graph STATIC src/native_graph.cpp)
target_include_directories(mas_native_graph PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(mas_native_graph PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(mas_native_graph PRIVATE /W4 /WX)
else()
  target_compile_options(mas_native_graph PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(mas_native_graph_test tests/native_graph_test.cpp)
target_link_libraries(mas_native_graph_test PRIVATE mas_native_graph Threads::Threads)
add_test(NAME native_graph_authorized_projection COMMAND mas_native_graph_test)

# N2a host-internal dispatcher. Graph + dispatcher remain independent of GUI,
# Node and CppMicroServices; callable only from a trusted native host.
add_library(mas_native_dispatch STATIC src/native_dispatch.cpp)
target_include_directories(mas_native_dispatch PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(mas_native_dispatch PUBLIC mas_native_graph Threads::Threads)
target_compile_features(mas_native_dispatch PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(mas_native_dispatch PRIVATE /W4 /WX)
else()
  target_compile_options(mas_native_dispatch PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(mas_native_dispatch_test tests/native_dispatch_test.cpp)
target_link_libraries(mas_native_dispatch_test PRIVATE mas_native_dispatch)
add_test(NAME native_dispatch_guarded_invoke COMMAND mas_native_dispatch_test)
